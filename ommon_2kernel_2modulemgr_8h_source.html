39/*
40 * Assigning the following macro to the variable sceKernelPreloadModuleInhibit with the OR operator inhibit preloading that module.
41 *
42 * Example
43 * <code>
44 * // Inhibit preload SceLibc and SceShellSvc.
45 * int sceKernelPreloadModuleInhibit = SCE_KERNEL_PRELOAD_INHIBIT_LIBC | SCE_KERNEL_PRELOAD_INHIBIT_LIBSHELLSVC;
46 * </code>
47 *
48 * And these are only valid for modules in the process image, preload is not inhibited even if specified for modules to be loaded later.
49 *
50 * WARNING
51 * If SceLibNet etc. is loaded without SceShellSvc etc. loaded, an unintended system crash will occur.
52 */
53typedef enum SceKernelPreloadInhibit {
54 SCE_KERNEL_PRELOAD_INHIBIT_NONE = 0x00000000,
55 SCE_KERNEL_PRELOAD_INHIBIT_LIBC = 0x10000,
56 SCE_KERNEL_PRELOAD_INHIBIT_LIBDBG = 0x20000,
57 SCE_KERNEL_PRELOAD_INHIBIT_LIBSHELLSVC = 0x80000,
58 SCE_KERNEL_PRELOAD_INHIBIT_LIBCDLG = 0x100000,
59 SCE_KERNEL_PRELOAD_INHIBIT_LIBFIOS2 = 0x200000,
60 SCE_KERNEL_PRELOAD_INHIBIT_APPUTIL = 0x400000,
61 SCE_KERNEL_PRELOAD_INHIBIT_LIBSCEFT2 = 0x800000,
62 SCE_KERNEL_PRELOAD_INHIBIT_LIBPVF = 0x1000000,
63 SCE_KERNEL_PRELOAD_INHIBIT_LIBPERF = 0x2000000
64} SceKernelPreloadInhibit;
